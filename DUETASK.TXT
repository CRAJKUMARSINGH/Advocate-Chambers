# DRAWING 12: OPTION 2 — PLOT 1 LONGITUDINAL SECTION (Through Courtyard)
# ============================================================
def draw_o2_p1_section():
    doc = setup_doc(\"SHEET 12 / O2-SEC-P1\", \"OPTION 2 — LONGITUDINAL SECTION PLOT 1 (Through Courtyard 24'x24')\",
                    \"1/8\\\" = 1'-0\\\"\")
    msp = doc.modelspace()
    W = 90 * FT
    plinth = 1.5 * FT
    gf_ht = 11 * FT
    slab = 1 * FT
    # Heights from dim table: GF=0, FF slab top=12', 2F=24', Roof=36', Parapet=39'
    levels = {\"Plinth\": 0, \"GF FF\": plinth, \"FF Slab Top\": plinth + gf_ht + slab,
              \"2F Slab Top\": plinth + 2 * (gf_ht + slab), \"Roof Top\": plinth + 3 * (gf_ht + slab),
              \"Parapet\": plinth + 3 * (gf_ht + slab) + 3 * FT,
              \"OHT Top\": plinth + 3 * (gf_ht + slab) + 4.5 * FT}
    gy = levels[\"Plinth\"]
    # Ground line
    msp.add_line((-3 * FT, gy), (W + 3 * FT, gy), dxfattribs={\"layer\": \"A-SECT-CUT\", \"lineweight\": 50})
    # Plinth
    rect(msp, 0, gy, W, levels[\"GF FF\"], layer=\"A-WALL\")
    fill_rect(msp, 0, gy, W, levels[\"GF FF\"], layer=\"A-HATCH\", hatch=\"ANSI31\")
    # Walls (cut) up to parapet
    wall_top_y = levels[\"Roof Top\"]
    for wx in [0, W]:
        msp.add_line((wx, levels[\"GF FF\"]), (wx, wall_top_y), dxfattribs={\"layer\": \"A-WALL\", \"lineweight\": 60})
        fill_rect(msp, wx - 0.3 * FT, levels[\"GF FF\"], wx + 0.4 * FT, wall_top_y, layer=\"A-WALL-PATT\")
    # Slabs (GF slab is just floor, then FF slab at level 12', 2F at 24', Roof at 36')
    for slab_level_name in [\"FF Slab Top\", \"2F Slab Top\", \"Roof Top\"]:
        top_y = levels[slab_level_name]
        bot_y = top_y - slab
        rect(msp, 0, bot_y, W, top_y, layer=\"A-ROOF\")
        fill_rect(msp, 0, bot_y, W, top_y, layer=\"A-HATCH\", hatch=\"ANSI31\")
        text_msp(msp, slab_level_name + \" (RCC 150mm M25 + 1 ft Coba)\", W / 2, bot_y - 0.8 * FT, h=0.2 * FT, layer=\"A-TEXT\")
    # Courtyard void (double height 23 ft clear, spans GF + FF)
    cx1 = (W - 24 * FT) / 2
    cx2 = cx1 + 24 * FT
    c_y_bot = levels[\"GF FF\"]

    msp.add_lwpolyline([(cx1, c_y_bot), (cx1, levels[\"Parapet\"]), (cx2, levels[\"Parapet\"]), (cx2, c_y_bot)],
                       dxfattribs={\"layer\": \"A-JALI\", \"linetype\": \"DASHED\"})
    text_msp(msp, \"24'x24' OPEN-TO-SKY COURTYARD VOID\", (cx1 + cx2) / 2, levels[\"2F Slab Top\"] + 2 * FT, h=0.3 * FT,
             layer=\"A-TEXT\")
    text_msp(msp, \"(DOUBLE HEIGHT 23' CLEAR GF→2F) | Stepwell Fountain + Trees\",
             (cx1 + cx2) / 2, levels[\"FF Slab Top\"] + gf_ht / 2, h=0.22 * FT, layer=\"A-TEXT\")
    # Fountain symbol
    msp.add_circle(((cx1 + cx2) / 2, c_y_bot + 2 * FT), 2 * FT, dxfattribs={\"layer\": \"A-HATCH\"})
    # Ring corridor (6 ft) text
    text_msp(msp, \"6 ft RING CORRIDOR FLOOR\", (cx1 - 6 * FT / 2), levels[\"FF Slab Top\"] + gf_ht / 2 - 5 * FT,
             h=0.2 * FT, layer=\"A-TEXT\", rot=90.0)
    text_msp(msp, \"6 ft RING CORRIDOR FLOOR\", (cx2 + 6 * FT / 2), levels[\"FF Slab Top\"] + gf_ht / 2 - 5 * FT,
             h=0.2 * FT, layer=\"A-TEXT\", rot=90.0)
    # Bay labels
    text_msp(msp, \"GF BAYS 60 Nos (8 Prem+52 Open)\", (0 + cx1) / 2, levels[\"GF FF\"] + 5.5 * FT, h=0.28 * FT, layer=\"A-TEXT\")
    text_msp(msp, \"FF BAYS 60 Nos (12 Prem+48 Open) | SKY-BRIDGE\",
             (W + cx2) / 2, levels[\"FF Slab Top\"] + 5.5 * FT, h=0.28 * FT, layer=\"A-TEXT\")
    # Sky-bridge indicator (right side, FF level)
    sb_x, sb_w = W, 10 * FT
    sb_y1, sb_y2 = levels[\"FF Slab Top\"], levels[\"FF Slab Top\"] + 1 * FT
    rect(msp, sb_x, sb_y1 - 0.5 * FT, sb_x + sb_w, sb_y2 + 3 * FT, layer=\"A-ROOF\")
    text_msp(msp, \"SKY-BRIDGE → PLOT 2 FF (10' Wide)\", sb_x + sb_w / 2, levels[\"FF Slab Top\"] + 2 * FT, h=0.22 * FT,

    # Level markers (right side)
    lmx = W + 4 * FT

    text_msp(msp, \"TOTAL HT (Plinth → Parapet): 39'-0\\\" < 15m Local Body Limit ✓\",
             W / 2, levels[\"Parapet\"] + 1.5 * FT, h=0.35 * FT, layer=\"A-TEXT-TTL\")
